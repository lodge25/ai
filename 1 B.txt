import math

board = [" "] * 9

def display():
    for i in range(0, 9, 3):
        print(" | ".join(board[i:i+3]))
        if i < 6:
            print("--+---+--")

def winner(b):
    lines = [
        (0,1,2), (3,4,5), (6,7,8),
        (0,3,6), (1,4,7), (2,5,8),
        (0,4,8), (2,4,6)
    ]

    for a, c, d in lines:
        if b[a] == b[c] == b[d] != " ":
            return b[a]

    return None

def minimax(b, is_ai):
    result = winner(b)

    if result == "O":
        return 1
    if result == "X":
        return -1
    if " " not in b:
        return 0

    if is_ai:
        best = -math.inf

        for i in range(9):
            if b[i] == " ":
                b[i] = "O"
                score = minimax(b, False)
                b[i] = " "
                best = max(best, score)

        return best

    else:
        best = math.inf

        for i in range(9):
            if b[i] == " ":
                b[i] = "X"
                score = minimax(b, True)
                b[i] = " "
                best = min(best, score)

        return best

def ai_move():
    best_score = -math.inf
    move = None

    for i in range(9):
        if board[i] == " ":
            board[i] = "O"
            score = minimax(board, False)
            board[i] = " "

            if score > best_score:
                best_score = score
                move = i

    if move is not None:
        board[move] = "O"

# Main game
print("Tic-Tac-Toe")
print("You are X, AI is O")
print("Positions:")
print("1 | 2 | 3")
print("--+---+--")
print("4 | 5 | 6")
print("--+---+--")
print("7 | 8 | 9")

while True:
    display()

    # Human move
    try:
        pos = int(input("Enter position (1-9): ")) - 1

        if pos < 0 or pos > 8 or board[pos] != " ":
            print("Invalid position! Try again.")
            continue

    except ValueError:
        print("Please enter a number from 1 to 9.")
        continue

    board[pos] = "X"

    if winner(board) == "X":
        display()
        print("Result: Win! Human wins.")
        break

    if " " not in board:
        display()
        print("Result: Draw!")
        break

    # AI move
    ai_move()
    print("AI played:")
    display()

    if winner(board) == "O":
        print("Result: Lose! AI wins.")
        break

    if " " not in board:
        print("Result: Draw!")
        break
